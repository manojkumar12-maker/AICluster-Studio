## 4. Cleanup Script

```powershell
# scripts/clean.ps1 — AICluster v2.0 Repository Cleanup
# Dry-run with -WhatIf, destructive without

param(
    [switch]$WhatIf = $false,
    [switch]$Force = $false
)

function Remove-ItemSafe {
    param($Path)
    if (Test-Path $Path) {
        if ($WhatIf) {
            Write-Host "[WOULD DELETE] $Path" -ForegroundColor Yellow
        } else {
            Remove-Item -Path $Path -Recurse -Force -ErrorAction SilentlyContinue
            Write-Host "[DELETED] $Path" -ForegroundColor Red
        }
    }
}

function Clear-DirectorySafe {
    param($Path)
    if (Test-Path $Path) {
        if ($WhatIf) {
            Write-Host "[WOULD CLEAR] $Path" -ForegroundColor Yellow
        } else {
            Remove-Item -Path "$Path\*" -Recurse -Force -ErrorAction SilentlyContinue
            Write-Host "[CLEARED] $Path" -ForegroundColor DarkYellow
        }
    }
}

Write-Host "AICluster v2.0 Repository Cleanup" -ForegroundColor Cyan
Write-Host "=================================" -ForegroundColor Cyan
if ($WhatIf) { Write-Host "DRY RUN — No changes will be made`n" -ForegroundColor Yellow }

# ── Phase 1: Stray Artifacts ──
Write-Host "`nPhase 1: Stray Artifacts" -ForegroundColor Green
Remove-ItemSafe -Path "nul"

# ── Phase 2: Test Build Artifacts ──
Write-Host "`nPhase 2: Test Build Artifacts" -ForegroundColor Green
Remove-ItemSafe -Path "build/hello"
Remove-ItemSafe -Path "build/hello2"
Remove-ItemSafe -Path "build/main_master"
Remove-ItemSafe -Path "build/test_pkg"
Remove-ItemSafe -Path "build/test_spec"

# ── Phase 3: Build Intermediates ──
Write-Host "`nPhase 3: Build Intermediates" -ForegroundColor Green
Remove-ItemSafe -Path "build/modules/build"
Remove-ItemSafe -Path "build/modules/dist"
Remove-ItemSafe -Path "build/modules/AIClusterMaster.spec"
Remove-ItemSafe -Path "build/modules/AIClusterWorker.spec"
Remove-ItemSafe -Path "build/modules/cli_entry_cli.spec"
Remove-ItemSafe -Path "build/modules/master_entry_master_version.txt"
Remove-ItemSafe -Path "build/modules/worker_entry_worker_version.txt"
Remove-ItemSafe -Path "build/modules/cli_entry_cli_version.txt"
Remove-ItemSafe -Path "backend/app/build"
Remove-ItemSafe -Path "backend/app/main_master_version.txt"
Remove-ItemSafe -Path "backend/app/dist"

# ── Phase 4: Duplicate EXE Directories ──
Write-Host "`nPhase 4: Duplicate EXE Directories" -ForegroundColor Green
Remove-ItemSafe -Path "dist/master"
Remove-ItemSafe -Path "dist/worker"
Remove-ItemSafe -Path "dist/aicluster.exe"
Remove-ItemSafe -Path "release"

# ── Phase 5: Python Bytecode ──
Write-Host "`nPhase 5: Python Bytecode Cache" -ForegroundColor Green
Get-ChildItem -Path "." -Directory -Filter "__pycache__" -Recurse -ErrorAction SilentlyContinue |
    ForEach-Object {
        if ($WhatIf) {
            Write-Host "[WOULD DELETE] $($_.FullName)" -ForegroundColor Yellow
        } else {
            Remove-Item -Path $_.FullName -Recurse -Force -ErrorAction SilentlyContinue
        }
    }
if (-not $WhatIf) { Write-Host "[DELETED] All __pycache__ directories" }

# ── Phase 6: Test/Lint Caches ──
Write-Host "`nPhase 6: Test/Lint Caches" -ForegroundColor Green
Remove-ItemSafe -Path "backend/.pytest_cache"
Remove-ItemSafe -Path "backend/.ruff_cache"
Remove-ItemSafe -Path "worker/.pytest_cache"

# ── Phase 7: Frontend Build ──
Write-Host "`nPhase 7: Frontend Build Cache" -ForegroundColor Green
Remove-ItemSafe -Path "frontend/.next"

# ── Phase 8: Environment Files ──
Write-Host "`nPhase 8: Environment Files" -ForegroundColor Green
Remove-ItemSafe -Path "backend/.env"
Remove-ItemSafe -Path "worker/.env"
Remove-ItemSafe -Path "frontend/.env.local"
Remove-ItemSafe -Path "backend/.env.example"

# ── Phase 9: Runtime Logs ──
Write-Host "`nPhase 9: Runtime Logs" -ForegroundColor Green
Remove-ItemSafe -Path "logs"
Remove-ItemSafe -Path "backend/logs"

# ── Phase 10: Large Dependencies (Prompt) ──
Write-Host "`nPhase 10: Large Dependencies (requires confirmation)" -ForegroundColor Green
$cleanDeps = $Force -or (Read-Host "Delete virtual environments (.venv)? (y/N)") -eq "y"
if ($cleanDeps) {
    Get-ChildItem -Path "." -Directory -Filter ".venv" -Recurse -ErrorAction SilentlyContinue |
        ForEach-Object {
            if ($WhatIf) {
                Write-Host "[WOULD DELETE] $($_.FullName)" -ForegroundColor Yellow
            } else {
                Remove-Item -Path $_.FullName -Recurse -Force -ErrorAction SilentlyContinue
                Write-Host "[DELETED] $($_.FullName)" -ForegroundColor Red
            }
        }
}

$cleanRust = $Force -or (Read-Host "Delete Rust build caches (target/)? (y/N)") -eq "y"
if ($cleanRust) {
    Get-ChildItem -Path "." -Directory -Filter "target" -Recurse -ErrorAction SilentlyContinue |
        Where-Object { $_.FullName -like "*src-tauri*" } |
        ForEach-Object {
            if ($WhatIf) {
                Write-Host "[WOULD DELETE] $($_.FullName)" -ForegroundColor Yellow
            } else {
                Remove-Item -Path $_.FullName -Recurse -Force -ErrorAction SilentlyContinue
                Write-Host "[DELETED] $($_.FullName)" -ForegroundColor Red
            }
        }
}

$cleanNode = $Force -or (Read-Host "Delete node_modules directories? (y/N)") -eq "y"
if ($cleanNode) {
    Get-ChildItem -Path "." -Directory -Filter "node_modules" -Recurse -ErrorAction SilentlyContinue |
        ForEach-Object {
            if ($WhatIf) {
                Write-Host "[WOULD DELETE] $($_.FullName)" -ForegroundColor Yellow
            } else {
                Remove-Item -Path $_.FullName -Recurse -Force -ErrorAction SilentlyContinue
                Write-Host "[DELETED] $($_.FullName)" -ForegroundColor Red
            }
        }
}

Write-Host "`nCleanup complete!" -ForegroundColor Cyan
```

## 6. Post-Cleanup Verification

```powershell
# Verify cleanup success
$errors = @()

# Essential files that must still exist
$mustExist = @(
    "backend/app/main.py",
    "worker/app/main.py",
    "studio/src-tauri/src/main.rs",
    "build/build.py",
    "config/default.yaml",
    "shared/py/__init__.py",
    "README.md",
    "VERSION"
)

foreach ($file in $mustExist) {
    if (-not (Test-Path $file)) {
        $errors += "MISSING: $file"
    }
}

# Files that must NOT exist
$mustNotExist = @(
    "nul",
    "build/hello",
    "backend/app/build"
)

foreach ($file in $mustNotExist) {
    if (Test-Path $file) {
        $errors += "STILL EXISTS: $file"
    }
}

if ($errors.Count -eq 0) {
    Write-Host "All checks passed. Repository is clean." -ForegroundColor Green
} else {
    $errors | ForEach-Object { Write-Host "FAIL: $_" -ForegroundColor Red }
}
```
